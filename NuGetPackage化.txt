なのでnuget specで生成されたNpoiWrapper.nuspecに修正・追記する。
・projectUrlをコメントアウト
・tagsを適当に修正
・releaseNotesをコメントアウト
・dependenciesを追記して明示。内容は.csprojのPackageReferenceを転記。
<?xml version="1.0" encoding="utf-8"?>
<package >
  <metadata>
    <id>$id$</id>
    <version>$version$</version>
    <title>$title$</title>
    <authors>$author$</authors>
    <requireLicenseAcceptance>false</requireLicenseAcceptance>
    <license type="expression">MIT</license>
    <!-- <icon>icon.png</icon> -->
    <!-- <projectUrl>http://project_url_here_or_delete_this_line/</projectUrl> -->
    <description>$description$</description>
    <!-- <releaseNotes>Summary of changes made in this release of the package.</releaseNotes> -->
    <copyright>$copyright$</copyright>
    <tags>Developers NpoiWrapper</tags>
    <dependencies>
      <group targetFramework=".NETFramework4.8">
        <dependency id="log4net" version="2.0.17" />
        <dependency id="NPOI" version="2.7.0" />
      </group>
    </dependencies>
  </metadata>
</package>
